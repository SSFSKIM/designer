def entries(path):
    # Consume one top-level object member at a time. json.JSONDecoder validates each
    # complete value; exact trailing delimiters and final EOF are checked separately.
    decoder=json.JSONDecoder();buffer='';eof=False
    with gzip.open(path,'rt') as f:
        def refill():
            nonlocal buffer,eof
            chunk=f.read(1048576);eof=not chunk;buffer+=chunk
        def whitespace():
            nonlocal buffer
            while True:
                buffer=buffer.lstrip()
                if buffer or eof:return
                refill()
        def value():
            nonlocal buffer
            whitespace()
            while True:
                try:obj,end=decoder.raw_decode(buffer);buffer=buffer[end:];return obj
                except json.JSONDecodeError:
                    if eof:raise
                    refill()
        def token(expected):
            nonlocal buffer
            whitespace();assert buffer.startswith(expected),(expected,buffer[:80]);buffer=buffer[len(expected):]
        token('{');first=True
        while True:
            whitespace()
            if buffer.startswith('}'):
                token('}');whitespace();assert not buffer and eof;break
            if not first:token(',')
            key=value();assert isinstance(key,str);token(':');row=value();yield key,row;first=False

def witness(cell,member,part,reading,channel,value,**extra):
    return dict(cell=cell,member=member,part=part,repeat='median' if reading==0 else reading-1,
                channel='RGB'[channel],codes=value,**extra)

def keep(record,name,row):
    if record.get(name) is None or row['codes']>record[name]['codes']:record[name]=row
