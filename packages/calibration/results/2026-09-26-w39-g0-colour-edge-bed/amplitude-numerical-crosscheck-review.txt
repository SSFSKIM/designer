===== varpro.py SHA256 0eb7cf83c43f17189342b9b63e02458126a4e4ae23cabcf9e5f9758f36ae9a67 =====
"""Diagnostic only: the amplitude model's minimum over (edge, gains) by variable projection
(scipy BFGS over the three free gains at each edge, the level and profile solved exactly),
against preflight.fit_amplitude at the declared cap. Shift-branch and quantised data."""
import importlib.util, sys, tempfile
from pathlib import Path
import numpy as np
from scipy.optimize import minimize
E=Path('/Users/new/vitrea-w39/g0/packages/calibration/results/2026-09-26-w39-g0-colour-edge-bed')
spec=importlib.util.spec_from_file_location('tp',E/'test-preflight.py'); T=importlib.util.module_from_spec(spec); spec.loader.exec_module(T)
P=T.P
captured=[]
orig=P.fit_amplitude
def spy(coords,Y,attested,inward,knots,**k):
    r=orig(coords,Y,attested,inward,knots,**k); captured.append((coords,Y,attested,inward,knots,r)); return r
P.fit_amplitude=spy
for name in sys.argv[1:]:
    captured.clear()
    with tempfile.TemporaryDirectory() as tmp:
        T.build_root(Path(tmp),name); P.verdict(Path(tmp))
    for coords,Y,att,inward,knots,r in captured:
        Ys=np.vstack(Y); n=len(Y)
        def f(g,B):
            gg=np.concatenate([[1.0],g])
            A=np.vstack([np.column_stack([np.ones(len(coords)),gg[p]*B]) for p in range(n)])
            return P.solve(A,Ys)[1]
        best=None
        for e in np.arange(att[0]-1,att[0]+1+1e-9,1/16):
            B=P.basis(coords,e,-1 if inward<0 else 1,knots) if False else P.basis(coords,e,inward,knots)
            vals=[]
            for g0 in ([1,1,1],r['gains'][1:],[0.9,0.8,0.7],[1.1,1.2,1.3]):
                m=minimize(f,np.asarray(g0,float),args=(B,),method='BFGS',options=dict(gtol=1e-10,maxiter=2000))
                vals.append(m.fun)
            v=min(vals)
            if best is None or v<best[0]: best=(v,e)
        print(name,'pathEdge0=%.3f'%att[0],'fit_amplitude rss=%.6g edge=%.4f conv=%s'%(r['rss'],r['edge'],r['converged']),
              '| varpro grid-min rss=%.6g at edge %.4f'%best, '| rel gap %.2e'%((r['rss']-best[0])/best[0]),flush=True)
